cmake_minimum_required(VERSION 3.20)
project(visualization.starlight LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(STARLIGHT_BUILD_ADDON "Build the Kodi GLES add-on" ON)
include(CTest)

add_library(starlight_spectrum STATIC src/Spectrum.cpp)
set_target_properties(starlight_spectrum PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(starlight_spectrum PUBLIC src)
target_compile_options(starlight_spectrum PRIVATE -Wall -Wextra -Werror)

if(STARLIGHT_BUILD_ADDON)
  find_path(KODI_INCLUDE_DIR kodi/addon-instance/Visualization.h REQUIRED)
  find_path(GLES_INCLUDE_DIR GLES2/gl2.h REQUIRED)
  find_library(GLES_LIBRARY GLESv2 REQUIRED)
  find_package(Threads REQUIRED)
  add_library(visualization.starlight MODULE src/Starlight.cpp)
  set_target_properties(visualization.starlight PROPERTIES PREFIX "" CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
  target_include_directories(visualization.starlight SYSTEM PRIVATE ${KODI_INCLUDE_DIR} ${GLES_INCLUDE_DIR})
  target_compile_definitions(visualization.starlight PRIVATE
    BUILD_KODI_ADDON TARGET_POSIX TARGET_LINUX ADDON_INSTANCE_VERSION_VISUALIZATION_USED)
  target_compile_options(visualization.starlight PRIVATE -Wall -Wextra -Werror)
  target_link_libraries(visualization.starlight PRIVATE starlight_spectrum ${GLES_LIBRARY} Threads::Threads)
  target_link_options(visualization.starlight PRIVATE -Wl,--no-undefined)
  install(TARGETS visualization.starlight LIBRARY DESTINATION visualization.starlight)
  install(FILES addon.xml LICENSE.txt DESTINATION visualization.starlight)
  install(DIRECTORY resources DESTINATION visualization.starlight)
endif()

if(BUILD_TESTING)
  add_executable(spectrum_test tests/spectrum_test.cpp)
  target_link_libraries(spectrum_test PRIVATE starlight_spectrum)
  target_compile_options(spectrum_test PRIVATE -Wall -Wextra -Werror)
  add_test(NAME spectrum COMMAND spectrum_test)
  if(STARLIGHT_BUILD_ADDON)
    add_executable(lifecycle_test tests/lifecycle_test.cpp)
    target_include_directories(lifecycle_test SYSTEM PRIVATE ${KODI_INCLUDE_DIR})
    target_compile_definitions(lifecycle_test PRIVATE TARGET_POSIX TARGET_LINUX)
    target_compile_options(lifecycle_test PRIVATE -Wall -Wextra -Werror)
    target_link_libraries(lifecycle_test PRIVATE ${CMAKE_DL_LIBS} Threads::Threads)
    add_test(NAME lifecycle COMMAND lifecycle_test $<TARGET_FILE:visualization.starlight>)
  endif()
endif()
